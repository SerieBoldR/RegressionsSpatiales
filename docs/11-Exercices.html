  wmat[i, queen_nb[[i]]] <- 1
}

  NO2 ~ Pct0_14 + Pct_65 + Pct_Img + Pct_brevet + NivVieMed + 
        adjacency(1|oid),
  data = st_drop_geometry(LyonIris),
  adjMatrix = wmat,
  family = gaussian()
)

  qtl <- rowQuantiles(tab, probs = c(0.025, 0.975))

  return(test)
})
